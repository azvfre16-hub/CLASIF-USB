<script>

    /* --------------------------------
       DATOS INICIALES
    -------------------------------- */

    let currentSection = "dj";

    let tracks = JSON.parse(
        localStorage.getItem("musicClassifierTracks")
    ) || [

        {
            name: "Demo Techno",
            artist: "Mi Proyecto",
            style: "Techno",
            bpm: 128,
            key: "Am",
            energy: 4,
            section: "dj"
        },

        {
            name: "Melodic Idea",
            artist: "Ableton Project",
            style: "Melodic",
            bpm: 120,
            key: "Fm",
            energy: 3,
            section: "ableton"
        }

    ];


    /* --------------------------------
       CAMBIAR SECCIÓN
    -------------------------------- */

    function changeSection(section, button) {

        currentSection = section;

        document
            .querySelectorAll(".tab")
            .forEach(tab => tab.classList.remove("active"));

        button.classList.add("active");

        renderTracks();
    }


    /* --------------------------------
       MOSTRAR TEMAS
    -------------------------------- */

    function renderTracks() {

        const list = document.getElementById("trackList");

        const search =
            document
                .getElementById("search")
                .value
                .toLowerCase();

        const style =
            document
                .getElementById("styleFilter")
                .value;

        const bpmFilter =
            document
                .getElementById("bpmFilter")
                .value;

        const energy =
            document
                .getElementById("energyFilter")
                .value;


        const filtered = tracks.filter(track => {

            if (track.section !== currentSection) {
                return false;
            }

            if (
                search &&
                !track.name.toLowerCase().includes(search) &&
                !track.artist.toLowerCase().includes(search)
            ) {
                return false;
            }

            if (style && track.style !== style) {
                return false;
            }

            if (energy && track.energy != energy) {
                return false;
            }

            if (bpmFilter === "slow" && track.bpm >= 110) {
                return false;
            }

            if (
                bpmFilter === "medium" &&
                (track.bpm < 110 || track.bpm > 125)
            ) {
                return false;
            }

            if (
                bpmFilter === "fast" &&
                (track.bpm < 126 || track.bpm > 140)
            ) {
                return false;
            }

            if (bpmFilter === "veryfast" && track.bpm <= 140) {
                return false;
            }

            return true;

        });


        list.innerHTML = "";


        if (filtered.length === 0) {

            list.innerHTML = `
                <tr>
                    <td colspan="7"
                        style="text-align:center;color:#777;padding:30px;">
                        No hay temas en esta categoría.
                    </td>
                </tr>
            `;

            return;
        }


        filtered.forEach((track, index) => {

            const row = document.createElement("tr");

            row.innerHTML = `

                <td>
                    <strong>${track.name}</strong>
                </td>

                <td>
                    ${track.artist}
                </td>

                <td>
                    <span class="tag">
                        ${track.style}
                    </span>
                </td>

                <td>
                    <span class="bpm">
                        ${track.bpm}
                    </span>
                </td>

                <td>
                    ${track.key}
                </td>

                <td>
                    <span class="energy">
                        ${"★".repeat(track.energy)}
                        ${"☆".repeat(5 - track.energy)}
                    </span>
                </td>

                <td>
                    <button
                        class="danger"
                        onclick="deleteTrack(${tracks.indexOf(track)})">
                        ×
                    </button>
                </td>

            `;

            list.appendChild(row);

        });

    }


    /* --------------------------------
       MODAL
    -------------------------------- */

    function openModal() {

        document
            .getElementById("modal")
            .classList.add("show");

    }


    function closeModal() {

        document
            .getElementById("modal")
            .classList.remove("show");

    }


    /* --------------------------------
       AÑADIR TEMA
    -------------------------------- */

    function addTrack() {

        const name =
            document.getElementById("trackName").value.trim();

        const artist =
            document.getElementById("artist").value.trim();

        const style =
            document.getElementById("style").value;

        const bpm =
            Number(document.getElementById("bpm").value);

        const key =
            document.getElementById("key").value.trim();

        const energy =
            Number(document.getElementById("energy").value);


        if (!name) {

            alert("Introduce el nombre del tema.");

            return;

        }


        const newTrack = {

            name: name,

            artist: artist || "Sin artista",

            style: style,

            bpm: bpm || 0,

            key: key || "-",

            energy: energy,

            section: currentSection

        };


        tracks.push(newTrack);


        saveTracks();

        renderTracks();

        closeModal();


        document.getElementById("trackName").value = "";
        document.getElementById("artist").value = "";
        document.getElementById("bpm").value = "";
        document.getElementById("key").value = "";

    }


    /* --------------------------------
       ELIMINAR
    -------------------------------- */

    function deleteTrack(index) {

        if (
            !confirm("¿Eliminar este tema?")
        ) {
            return;
        }


        tracks.splice(index, 1);

        saveTracks();

        renderTracks();

    }


    /* --------------------------------
       GUARDAR
    -------------------------------- */

    function saveTracks() {

        localStorage.setItem(
            "musicClassifierTracks",
            JSON.stringify(tracks)
        );

    }


    /* --------------------------------
       LIMPIAR FILTROS
    -------------------------------- */

    function clearFilters() {

        document.getElementById("search").value = "";

        document.getElementById("styleFilter").value = "";

        document.getElementById("bpmFilter").value = "";

        document.getElementById("energyFilter").value = "";

        renderTracks();

    }


    /* --------------------------------
       INICIAR
    -------------------------------- */

    renderTracks();

</script>
